```sql
WITH event_budgets AS (
    SELECT link_to_event, SUM(amount) AS total_budgeted
    FROM main.budget
    GROUP BY link_to_event
),
event_attendance AS (
    SELECT link_to_event, COUNT(link_to_member) AS members_attended
    FROM main.attendance
    GROUP BY link_to_event
)
SELECT
    e.event_name,
    e.event_date,
    COALESCE(b.total_budgeted, 0) AS total_budgeted,
    COALESCE(a.members_attended, 0) AS members_attended
FROM main.event AS e
LEFT JOIN event_budgets AS b ON b.link_to_event = e.event_id
LEFT JOIN event_attendance AS a ON a.link_to_event = e.event_id
ORDER BY e.event_name ASC, e.event_date ASC
LIMIT -1
```
