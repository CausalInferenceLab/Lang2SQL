```sql
WITH RECURSIVE items(expense_id, item_name, remaining) AS (
    SELECT e.expense_id, NULL, e.expense_description || ','
    FROM main.expense AS e
    JOIN main.member AS m ON m.member_id = e.link_to_member
    WHERE m.first_name = 'Sacha' AND m.last_name = 'Harrison'

    UNION ALL

    SELECT expense_id,
           trim(substr(remaining, 1, instr(remaining, ',') - 1)),
           substr(remaining, instr(remaining, ',') + 1)
    FROM items
    WHERE remaining <> ''
),
named_items AS (
    SELECT expense_id,
           CASE WHEN item_name GLOB 'Bakery - *'
                THEN substr(item_name, length('Bakery - ') + 1)
                ELSE item_name
           END AS item_name
    FROM items
    WHERE item_name <> ''
)
SELECT item_name, COUNT(DISTINCT expense_id) AS purchase_count
FROM named_items
GROUP BY item_name
ORDER BY purchase_count DESC, item_name
LIMIT 1000
```
