```sql
SELECT ROUND(100.0 * AVG(CASE WHEN a.alignment = 'Good' THEN 1 ELSE 0 END), 2) AS good_percentage
FROM superhero AS s
JOIN publisher AS p ON p.id = s.publisher_id
LEFT JOIN alignment AS a ON a.id = s.alignment_id
WHERE p.publisher_name = 'Marvel Comics'
  AND TRIM(s.full_name) NOT IN ('', '-')
LIMIT 1
```
