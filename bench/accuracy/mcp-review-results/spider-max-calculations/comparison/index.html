WITH player_stint_highs AS (
    SELECT
        player_id,
        MAX(g) AS games_played,
        MAX(NULLIF(r, '')) AS runs,
        MAX(NULLIF(h, '')) AS hits,
        MAX(NULLIF(hr, '')) AS home_runs
    FROM main.batting
    GROUP BY player_id
),
scores (player_id, stat_order, statistic, score) AS (
    SELECT player_id, 1, 'Games played', games_played FROM player_stint_highs
    UNION ALL
    SELECT player_id, 2, 'Runs', runs FROM player_stint_highs
    UNION ALL
    SELECT player_id, 3, 'Hits', hits FROM player_stint_highs
    UNION ALL
    SELECT player_id, 4, 'Home runs', home_runs FROM player_stint_highs
),
ranked AS (
    SELECT scores.*, MAX(score) OVER (PARTITION BY statistic) AS record_score
    FROM scores
)
SELECT r.statistic, p.name_given, r.score
FROM ranked AS r
JOIN main.player AS p ON p.player_id = r.player_id
WHERE r.score = r.record_score
ORDER BY r.stat_order, p.name_given
LIMIT -1

WITH role_counts AS (
    SELECT player_id, role, COUNT(*) AS matches_in_role
    FROM main.player_match
    GROUP BY player_id, role
),
ranked_roles AS (
    SELECT player_id, role,
           SUM(matches_in_role) OVER (PARTITION BY player_id) AS matches_played,
           ROW_NUMBER() OVER (
               PARTITION BY player_id ORDER BY matches_in_role DESC, role
           ) AS role_rank
    FROM role_counts
),
deliveries AS (
    SELECT b.match_id, b.striker, b.bowler,
           COALESCE(s.runs_scored, 0) AS runs,
           CASE WHEN w.kind_out IN (
               'bowled', 'caught', 'caught and bowled', 'hit wicket', 'lbw', 'stumped'
           ) THEN 1 ELSE 0 END AS bowler_wicket
    FROM main.ball_by_ball AS b
    LEFT JOIN main.batsman_scored AS s USING (match_id, over_id, ball_id, innings_no)
    LEFT JOIN main.wicket_taken AS w USING (match_id, over_id, ball_id, innings_no)
),
batting_by_match AS (
    SELECT striker AS player_id, match_id, SUM(runs) AS match_runs,
           COUNT(*) AS balls_faced
    FROM deliveries
    GROUP BY striker, match_id
),
batting AS (
    SELECT player_id, SUM(match_runs) AS total_runs,
           MAX(match_runs) AS highest_score,
           SUM(match_runs >= 30) AS matches_30_plus,
           SUM(match_runs >= 50) AS matches_50_plus,
           SUM(match_runs >= 100) AS matches_100_plus,
           SUM(balls_faced) AS total_balls_faced
    FROM batting_by_match
    GROUP BY player_id
),
dismissals AS (
    SELECT player_out AS player_id, COUNT(*) AS total_dismissals
    FROM main.wicket_taken
    WHERE kind_out <> 'retired hurt'
    GROUP BY player_out
),
bowling_by_match AS (
    SELECT bowler AS player_id, match_id, SUM(bowler_wicket) AS wickets,
           SUM(runs) AS runs_given, COUNT(*) AS balls_bowled
    FROM deliveries
    GROUP BY bowler, match_id
),
bowling AS (
    SELECT player_id, wickets, runs_given,
           SUM(wickets) OVER (PARTITION BY player_id) AS total_wickets,
           6.0 * SUM(runs_given) OVER (PARTITION BY player_id)
               / SUM(balls_bowled) OVER (PARTITION BY player_id) AS economy_rate,
           ROW_NUMBER() OVER (
               PARTITION BY player_id ORDER BY wickets DESC, runs_given, match_id
           ) AS performance_rank
    FROM bowling_by_match
)
SELECT p.player_id, p.player_name, r.role AS most_frequent_role,
       p.batting_hand, p.bowling_skill,
       COALESCE(b.total_runs, 0) AS total_runs,
       COALESCE(r.matches_played, 0) AS total_matches_played,
       COALESCE(d.total_dismissals, 0) AS total_dismissals,
       1.0 * COALESCE(b.total_runs, 0) / NULLIF(d.total_dismissals, 0) AS batting_average,
       COALESCE(b.highest_score, 0) AS highest_score,
       COALESCE(b.matches_30_plus, 0) AS matches_30_plus,
       COALESCE(b.matches_50_plus, 0) AS matches_50_plus,
       COALESCE(b.matches_100_plus, 0) AS matches_100_plus,
       COALESCE(b.total_balls_faced, 0) AS total_balls_faced,
       100.0 * b.total_runs / NULLIF(b.total_balls_faced, 0) AS strike_rate,
       COALESCE(w.total_wickets, 0) AS total_wickets,
       w.economy_rate,
       w.wickets || '-' || w.runs_given AS best_bowling_performance
FROM main.player AS p
LEFT JOIN ranked_roles AS r ON r.player_id = p.player_id AND r.role_rank = 1
LEFT JOIN batting AS b ON b.player_id = p.player_id
LEFT JOIN dismissals AS d ON d.player_id = p.player_id
LEFT JOIN bowling AS w ON w.player_id = p.player_id AND w.performance_rank = 1
ORDER BY p.player_id
LIMIT 1000

WITH role_stats AS (
    SELECT pm.player_id, pm.role,
           SUM(COUNT(*)) OVER (PARTITION BY pm.player_id) AS matches_played,
           ROW_NUMBER() OVER (
               PARTITION BY pm.player_id
               -- Choose the alphabetically first role when match counts tie.
               ORDER BY COUNT(*) DESC, pm.role ASC
           ) AS role_rank
    FROM main.player_match AS pm
    GROUP BY pm.player_id, pm.role
),
deliveries AS (
    -- Count every recorded delivery and take runs from the batting score.
    SELECT b.match_id, b.striker, b.bowler,
           COALESCE(s.runs_scored, 0) AS runs,
           CASE WHEN w.kind_out IN (
               'bowled', 'caught', 'caught and bowled', 'hit wicket', 'lbw', 'stumped'
           ) THEN 1 ELSE 0 END AS bowler_wicket
    FROM main.ball_by_ball AS b
    LEFT JOIN main.batsman_scored AS s USING (match_id, innings_no, over_id, ball_id)
    LEFT JOIN main.wicket_taken AS w USING (match_id, innings_no, over_id, ball_id)
),
batting_by_match AS (
    SELECT player_id, match_id, SUM(runs) AS match_runs, SUM(balls_faced) AS balls_faced
    FROM (
        SELECT striker AS player_id, match_id, runs, 1 AS balls_faced
        FROM deliveries
        UNION ALL
        -- Include batters who appeared only at the non-striker's end.
        SELECT DISTINCT non_striker, match_id, 0, 0
        FROM main.ball_by_ball
    ) AS batters
    GROUP BY player_id, match_id
),
batting AS (
    SELECT player_id, SUM(match_runs) AS total_runs, MAX(match_runs) AS highest_match_score,
           COUNT(*) FILTER (WHERE match_runs >= 30) AS matches_30_plus,
           COUNT(*) FILTER (WHERE match_runs >= 50) AS matches_50_plus,
           COUNT(*) FILTER (WHERE match_runs >= 100) AS matches_100_plus,
           SUM(balls_faced) AS total_balls_faced
    FROM batting_by_match
    GROUP BY player_id
),
dismissals AS (
    SELECT player_out AS player_id, COUNT(*) AS total_dismissals
    FROM main.wicket_taken
    WHERE kind_out <> 'retired hurt'
    GROUP BY player_out
),
bowling_by_match AS (
    SELECT bowler AS player_id, match_id, SUM(bowler_wicket) AS wickets,
           SUM(runs) AS runs_given, COUNT(*) AS balls_bowled
    FROM deliveries
    GROUP BY bowler, match_id
),
bowling AS (
    SELECT player_id, wickets, runs_given,
           SUM(wickets) OVER career AS total_wickets,
           6.0 * SUM(runs_given) OVER career
               / NULLIF(SUM(balls_bowled) OVER career, 0) AS economy_rate,
           ROW_NUMBER() OVER (
               career ORDER BY wickets DESC, runs_given ASC, match_id ASC
           ) AS performance_rank
    FROM bowling_by_match
    WINDOW career AS (PARTITION BY player_id)
)
SELECT p.player_id, p.player_name, r.role AS most_frequent_role,
       p.batting_hand, p.bowling_skill,
       COALESCE(bt.total_runs, 0) AS total_runs_scored,
       COALESCE(r.matches_played, 0) AS total_matches_played,
       COALESCE(d.total_dismissals, 0) AS total_dismissals,
       1.0 * COALESCE(bt.total_runs, 0) / NULLIF(d.total_dismissals, 0) AS batting_average,
       bt.highest_match_score,
       COALESCE(bt.matches_30_plus, 0) AS matches_30_plus,
       COALESCE(bt.matches_50_plus, 0) AS matches_50_plus,
       COALESCE(bt.matches_100_plus, 0) AS matches_100_plus,
       COALESCE(bt.total_balls_faced, 0) AS total_balls_faced,
       100.0 * bt.total_runs / NULLIF(bt.total_balls_faced, 0) AS strike_rate,
       COALESCE(bw.total_wickets, 0) AS total_wickets_taken,
       bw.economy_rate,
       bw.wickets || '-' || bw.runs_given AS best_bowling_performance
-- Match records for ID 35 lack a corresponding player profile.
FROM main.player AS p
LEFT JOIN role_stats AS r ON r.player_id = p.player_id AND r.role_rank = 1
LEFT JOIN batting AS bt ON bt.player_id = p.player_id
LEFT JOIN dismissals AS d ON d.player_id = p.player_id
LEFT JOIN bowling AS bw ON bw.player_id = p.player_id AND bw.performance_rank = 1
ORDER BY p.player_id
LIMIT -1

WITH RECURSIVE
daily_changes AS (
    SELECT customer_id, txn_date,
           SUM(CASE WHEN txn_type = 'deposit' THEN txn_amount ELSE -txn_amount END) AS net_change
    FROM main.customer_transactions
    GROUP BY customer_id, txn_date
),
customer_days(customer_id, balance_date, baseline_month) AS (
    SELECT customer_id, MIN(txn_date), strftime('%Y-%m', MIN(txn_date))
    FROM daily_changes
    GROUP BY customer_id

    UNION ALL

    SELECT customer_id, date(balance_date, '+1 day'), baseline_month
    FROM customer_days
    WHERE balance_date < (SELECT MAX(txn_date) FROM daily_changes)
),
daily_balances AS (
    SELECT days.customer_id, days.balance_date, days.baseline_month,
           SUM(COALESCE(changes.net_change, 0)) OVER (
               PARTITION BY days.customer_id
               ORDER BY days.balance_date
               ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
           ) AS closing_balance
    FROM customer_days AS days
    LEFT JOIN daily_changes AS changes
        ON changes.customer_id = days.customer_id
       AND changes.txn_date = days.balance_date
),
rolling_averages AS (
    SELECT customer_id, strftime('%Y-%m', balance_date) AS month, baseline_month,
           AVG(closing_balance) OVER trailing_30_days AS avg_balance,
           COUNT(*) OVER trailing_30_days AS days_in_window
    FROM daily_balances
    WINDOW trailing_30_days AS (
        PARTITION BY customer_id
        ORDER BY balance_date
        ROWS BETWEEN 29 PRECEDING AND CURRENT ROW
    )
),
customer_monthly_maxima AS (
    SELECT customer_id, month,
           MAX(CASE WHEN avg_balance < 0 THEN 0.0 ELSE avg_balance END) AS max_avg_balance
    FROM rolling_averages
    WHERE days_in_window = 30 AND month > baseline_month
    GROUP BY customer_id, month
)
SELECT month, SUM(max_avg_balance) AS total_max_30_day_avg_balance
FROM customer_monthly_maxima
GROUP BY month
ORDER BY month
LIMIT 1000

WITH scoring AS (
    SELECT match_id, innings_no, over_id, runs_scored AS runs
    FROM main.batsman_scored
    UNION ALL
    SELECT match_id, innings_no, over_id, extra_runs AS runs
    FROM main.extra_runs
),
ranked_overs AS (
    SELECT match_id, innings_no, over_id, SUM(runs) AS total_runs,
           ROW_NUMBER() OVER (
               PARTITION BY match_id
               -- Select the earliest over when totals tie.
               ORDER BY SUM(runs) DESC, innings_no, over_id
           ) AS over_rank
    FROM scoring
    GROUP BY match_id, innings_no, over_id
),
winning_over_bowlers AS (
    SELECT r.match_id, r.innings_no, r.over_id, r.total_runs,
           b.bowler, MIN(b.ball_id) AS first_ball
    FROM ranked_overs AS r
    LEFT JOIN main.ball_by_ball AS b USING (match_id, innings_no, over_id)
    WHERE r.over_rank = 1
    GROUP BY r.match_id, r.innings_no, r.over_id, r.total_runs, b.bowler
)
SELECT match_id, innings_no, over_id, total_runs,
       GROUP_CONCAT(bowler, ', ' ORDER BY first_ball) AS bowler_ids,
       AVG(total_runs) OVER () AS avg_highest_over_runs
FROM winning_over_bowlers
GROUP BY match_id, innings_no, over_id, total_runs
ORDER BY match_id
LIMIT -1
