```sql
SELECT
    e.event_name,
    e.location,
    COUNT(a.link_to_member) AS members_attended
FROM main.event AS e
LEFT JOIN main.attendance AS a
    ON a.link_to_event = e.event_id
WHERE e.type = 'Budget'
GROUP BY e.event_id, e.event_name, e.location
ORDER BY e.event_date, e.event_id
LIMIT -1
```
