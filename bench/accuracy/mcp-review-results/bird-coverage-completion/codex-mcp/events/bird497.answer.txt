```sql
WITH atom_bonds AS (
    SELECT atom_id, bond_id FROM main.connected
    UNION
    SELECT atom_id2, bond_id FROM main.connected
),
single_bond_only AS (
    SELECT ab.atom_id
    FROM atom_bonds AS ab
    LEFT JOIN main.bond AS b ON b.bond_id = ab.bond_id
    GROUP BY ab.atom_id
    HAVING COUNT(*) = COUNT(CASE WHEN b.bond_type = '-' THEN 1 END)
)
SELECT a.element, COUNT(s.atom_id) AS single_bond_only_atoms
FROM main.atom AS a
LEFT JOIN single_bond_only AS s ON s.atom_id = a.atom_id
GROUP BY a.element
ORDER BY a.element
LIMIT -1
```
