```sql
SELECT ROUND(
    100.0 * SUM(CASE WHEN res.position = 1 THEN 0 ELSE 1 END)
    / NULLIF(COUNT(*), 0), 2
) AS non_victory_percentage
FROM results AS res
JOIN drivers AS d ON d.driverId = res.driverId
JOIN races AS r ON r.raceId = res.raceId
WHERE d.forename = 'Lewis'
  AND d.surname = 'Hamilton'
  AND r.year >= 2010
LIMIT 1000
```
