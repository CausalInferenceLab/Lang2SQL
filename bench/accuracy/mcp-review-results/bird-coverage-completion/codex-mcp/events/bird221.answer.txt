```sql
WITH district_withdrawals AS (
    SELECT a.district_id, SUM(t.amount) AS total_withdrawn
    FROM main.trans AS t
    JOIN main.account AS a ON a.account_id = t.account_id
    WHERE t.type = 'VYDAJ'
    GROUP BY a.district_id
),
account_orders AS (
    SELECT o.account_id,
           SUM(CAST(ROUND(o.amount * 10) AS INTEGER)) AS debit_tenths
    FROM main."order" AS o
    GROUP BY o.account_id
),
district_orders AS (
    SELECT a.district_id,
           SUM(COALESCE(o.debit_tenths, 0)) * 10 AS debit_hundredths,
           COUNT(*) AS account_count
    FROM main.account AS a
    LEFT JOIN account_orders AS o ON o.account_id = a.account_id
    GROUP BY a.district_id
),
district_metrics AS (
    SELECT d.A2 AS district_name,
           d.A3 AS region,
           CAST(d.A4 AS INTEGER) AS population,
           w.total_withdrawn,
           1.0 * w.total_withdrawn / CAST(d.A4 AS INTEGER) AS per_inhabitant,
           (s.debit_hundredths / s.account_count
            + CASE WHEN 2 * ABS(s.debit_hundredths % s.account_count) >= s.account_count
                   THEN SIGN(s.debit_hundredths) ELSE 0 END
           ) / 100.0 AS avg_standing_order_debit_per_account
    FROM main.district AS d
    JOIN district_withdrawals AS w ON w.district_id = d.district_id
    JOIN district_orders AS s ON s.district_id = d.district_id
    WHERE CAST(d.A4 AS INTEGER) >= 100000
)
SELECT district_name,
       region,
       population,
       total_withdrawn,
       ROUND(per_inhabitant, 2) AS withdrawals_per_inhabitant,
       RANK() OVER (ORDER BY per_inhabitant DESC) AS withdrawal_rank,
       avg_standing_order_debit_per_account
FROM district_metrics
ORDER BY withdrawal_rank, district_name
LIMIT -1
```
