```sql
WITH annual_totals AS (
    SELECT c.Segment, c.CustomerID,
           SUM(CAST(ROUND(y.Consumption * 100) AS INTEGER)) AS total_hundredths
    FROM main.customers AS c
    JOIN main.yearmonth AS y ON y.CustomerID = c.CustomerID
    WHERE c.Currency = 'CZK'
      AND c.Segment IN ('SME', 'LAM', 'KAM')
      AND y.Date BETWEEN '201301' AND '201312'
      AND y.Consumption IS NOT NULL
    GROUP BY c.Segment, c.CustomerID
),
ranked AS (
    SELECT annual_totals.*,
           ROW_NUMBER() OVER (
               PARTITION BY Segment ORDER BY total_hundredths, CustomerID
           ) AS segment_rank
    FROM annual_totals
),
winners AS (
    SELECT Segment, CustomerID, total_hundredths
    FROM ranked
    WHERE segment_rank = 1
)
SELECT sme.CustomerID AS sme_customer_id, sme.total_hundredths / 100.0 AS sme_total,
       lam.CustomerID AS lam_customer_id, lam.total_hundredths / 100.0 AS lam_total,
       kam.CustomerID AS kam_customer_id, kam.total_hundredths / 100.0 AS kam_total,
       ROUND((sme.total_hundredths - lam.total_hundredths) / 100.0, 2) AS sme_minus_lam,
       ROUND((lam.total_hundredths - kam.total_hundredths) / 100.0, 2) AS lam_minus_kam,
       ROUND((kam.total_hundredths - sme.total_hundredths) / 100.0, 2) AS kam_minus_sme
FROM winners AS sme
CROSS JOIN winners AS lam
CROSS JOIN winners AS kam
WHERE sme.Segment = 'SME' AND lam.Segment = 'LAM' AND kam.Segment = 'KAM'
LIMIT 1
```
