```sql
SELECT ROUND(
    100.0 * COUNT(CASE WHEN p.SEX = 'F' THEN 1 END) / NULLIF(COUNT(*), 0),
    1
) AS women_percentage
FROM main.Patient AS p
WHERE p.Birthday >= '1936-01-01'
  AND p.Birthday < '1957-01-01'
  AND p.SEX IN ('F', 'M')
  AND NOT EXISTS (
      SELECT 1
      FROM main.Laboratory AS l
      WHERE l.ID = p.ID
        AND l.CPK >= 250
  )
LIMIT 1
```
