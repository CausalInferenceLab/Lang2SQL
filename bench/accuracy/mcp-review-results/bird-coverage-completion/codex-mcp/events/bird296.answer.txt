```sql
SELECT
    SUM(b.spent) FILTER (WHERE e.event_date < '2020-01-01')
    - SUM(b.spent) FILTER (WHERE e.event_date >= '2020-01-01') AS spent_2019_minus_2020
FROM main.budget AS b
JOIN main.event AS e ON e.event_id = b.link_to_event
WHERE e.event_date >= '2019-01-01'
  AND e.event_date < '2021-01-01'
LIMIT 1
```
