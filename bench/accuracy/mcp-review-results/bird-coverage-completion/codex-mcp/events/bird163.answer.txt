```sql
WITH team_wins AS (
    SELECT m.away_team_api_id, COUNT(*) AS away_wins
    FROM main."Match" AS m
    JOIN main.League AS l ON l.id = m.league_id
    WHERE l.name = 'Scotland Premier League'
      AND m.season = '2009/2010'
      AND m.away_team_goal > m.home_team_goal
    GROUP BY m.away_team_api_id
)
SELECT t.team_long_name, w.away_wins
FROM team_wins AS w
JOIN main.Team AS t ON t.team_api_id = w.away_team_api_id
WHERE w.away_wins = (SELECT MAX(away_wins) FROM team_wins)
ORDER BY t.team_long_name
LIMIT 1000
```
