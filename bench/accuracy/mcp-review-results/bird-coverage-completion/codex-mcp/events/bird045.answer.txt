```sql
SELECT s.Street, s.City, s.Zip, s.State
FROM main.satscores AS t
JOIN main.schools AS s ON s.CDSCode = t.cds
JOIN main.frpm AS f ON f.CDSCode = t.cds
WHERE t.rtype = 'S'
  AND t.NumTstTakr >= 700
  AND t.NumGE1500 IS NOT NULL
ORDER BY CAST(t.NumGE1500 AS REAL) / t.NumTstTakr ASC,
         CAST(f."School Code" AS INTEGER) ASC
LIMIT 1
```
