```sql
SELECT DISTINCT s.name, s.releaseDate
FROM cards AS c
JOIN sets AS s ON s.code = c.setCode
WHERE c.name = 'Tarmogoyf'
  AND c.isReprint = 1
  AND c.frameVersion GLOB '[0-9][0-9][0-9][0-9]'
  AND CAST(strftime('%Y', s.releaseDate) AS INTEGER) - CAST(c.frameVersion AS INTEGER) >= 10
ORDER BY s.releaseDate, s.name
LIMIT 1000
```
