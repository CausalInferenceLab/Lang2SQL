```sql
SELECT
    ROUND(AVG(NULLIF(s.height_cm, 0)), 2) AS average_height_cm,
    ROUND(AVG(NULLIF(s.weight_kg, 0)), 2) AS average_weight_kg
FROM superhero AS s
JOIN alignment AS a ON a.id = s.alignment_id
JOIN colour AS c ON c.id = s.skin_colour_id
WHERE a.alignment = 'Bad' AND c.colour = 'Green'
LIMIT 1
```
