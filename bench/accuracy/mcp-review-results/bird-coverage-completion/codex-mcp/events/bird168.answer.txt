```sql
SELECT
    AVG(a.ball_control) FILTER (WHERE p.player_name = 'Abdou Diallo')
    - AVG(a.ball_control) FILTER (WHERE p.player_name = 'Aaron Appindangoye')
        AS mean_ball_control_difference
FROM main.Player AS p
JOIN main.Player_Attributes AS a ON a.player_api_id = p.player_api_id
WHERE p.player_name IN ('Abdou Diallo', 'Aaron Appindangoye')
LIMIT 1000
```
