WITH sessions AS (
    SELECT session,
           MIN(CASE WHEN path IN ('/detail', '/complete') THEN stamp END) AS first_target
    FROM main.activity_log

),
counted_events AS (
    SELECT s.session, e.path, e.search_type,
           COUNT(e.search_type) OVER (PARTITION BY s.session) AS event_count
    FROM sessions AS s
    LEFT JOIN main.activity_log AS e
      ON e.session = s.session
     AND e.search_type <> ''
     AND (s.first_target IS NULL OR e.stamp < s.first_target)
)
SELECT DISTINCT session, path, search_type, event_count
FROM counted_events
WHERE event_count = (SELECT MIN(event_count) FROM counted_events)
ORDER BY session, path, search_type
LIMIT 1000

WITH excluded_laps AS (
    SELECT race_id, driver_id, lap FROM main.pit_stops
    UNION
    SELECT race_id, driver_id, lap + 1 FROM main.pit_stops
    UNION
    SELECT race_id, driver_id, lap FROM main.retirements
),
eligible_laps AS MATERIALIZED (
    SELECT cur.race_id, cur.lap, cur.driver_id,
           prev.position AS previous_position, cur.position
    FROM main.lap_times AS cur
    JOIN main.lap_times AS prev
      ON prev.race_id = cur.race_id
     AND prev.driver_id = cur.driver_id
     AND prev.lap = cur.lap - 1
    JOIN main.races_ext AS r ON r.race_id = cur.race_id
    LEFT JOIN excluded_laps AS x
      ON x.race_id = cur.race_id
     AND x.driver_id = cur.driver_id
     AND x.lap = cur.lap
    WHERE cur.lap > 1
      AND r.is_pit_data_available = 1
      AND x.driver_id IS NULL
),
passes AS MATERIALIZED (
    SELECT a.driver_id AS passer, b.driver_id AS passed
    FROM eligible_laps AS a
    JOIN eligible_laps AS b
      ON b.race_id = a.race_id
     AND b.lap = a.lap
     AND a.previous_position > b.previous_position
     AND a.position < b.position
),
driver_events AS (
    SELECT passed AS driver_id, 1 AS balance FROM passes
    UNION ALL
    SELECT passer AS driver_id, -1 AS balance FROM passes
)
SELECT d.full_name
FROM main.drivers_ext AS d
JOIN driver_events AS e ON e.driver_id = d.driver_id
GROUP BY d.driver_id, d.full_name
HAVING SUM(e.balance) > 0
ORDER BY d.full_name
LIMIT 1000

WITH match_winners AS (
    SELECT CASE
               WHEN m.home_team_goal > m.away_team_goal THEN m.home_team_api_id
               ELSE m.away_team_api_id
           END AS team_api_id
    FROM main."Match" AS m
    WHERE m.home_team_goal <> m.away_team_goal
)
SELECT t.team_long_name AS team, COUNT(*) AS wins
FROM match_winners AS w
JOIN main.Team AS t ON t.team_api_id = w.team_api_id
GROUP BY t.team_api_id, t.team_long_name
ORDER BY wins DESC, t.team_long_name, t.team_api_id
LIMIT 10;
