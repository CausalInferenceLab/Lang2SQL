WITH team_lineups AS (
    SELECT id AS match_id,
           CASE WHEN home_team_goal > away_team_goal THEN 'Win' ELSE 'Loss' END AS outcome,
           json_array(
               home_player_1, home_player_2, home_player_3, home_player_4,
               home_player_5, home_player_6, home_player_7, home_player_8,
               home_player_9, home_player_10, home_player_11
           ) AS lineup
    FROM main."Match"
    WHERE home_team_goal <> away_team_goal

    UNION ALL

    SELECT id AS match_id,
           CASE WHEN away_team_goal > home_team_goal THEN 'Win' ELSE 'Loss' END AS outcome,
           json_array(
               away_player_1, away_player_2, away_player_3, away_player_4,
               away_player_5, away_player_6, away_player_7, away_player_8,
               away_player_9, away_player_10, away_player_11
           ) AS lineup
    FROM main."Match"
    WHERE home_team_goal <> away_team_goal
),
player_totals AS (
    SELECT t.outcome,
           slot.value AS player_api_id,
           COUNT(DISTINCT t.match_id) AS match_count
    FROM team_lineups AS t
    CROSS JOIN json_each(t.lineup) AS slot
    WHERE slot.value IS NOT NULL
    GROUP BY t.outcome, slot.value
),
ranked AS (
    SELECT outcome, player_api_id, match_count,
           RANK() OVER (PARTITION BY outcome ORDER BY match_count DESC) AS outcome_rank
    FROM player_totals
)
SELECT r.outcome, p.player_name, r.match_count
FROM ranked AS r
JOIN main.Player AS p ON p.player_api_id = r.player_api_id
WHERE r.outcome_rank = 1
ORDER BY CASE r.outcome WHEN 'Win' THEN 1 ELSE 2 END, p.player_name
LIMIT 1000

WITH team_lineups AS (
    SELECT match_api_id,
           CASE WHEN home_team_goal > away_team_goal THEN 'Win' ELSE 'Loss' END AS outcome,
           json_array(
               home_player_1, home_player_2, home_player_3, home_player_4,
               home_player_5, home_player_6, home_player_7, home_player_8,
               home_player_9, home_player_10, home_player_11
           ) AS lineup
    FROM main."Match"
    WHERE home_team_goal <> away_team_goal

    UNION ALL

    SELECT match_api_id,
           CASE WHEN away_team_goal > home_team_goal THEN 'Win' ELSE 'Loss' END AS outcome,
           json_array(
               away_player_1, away_player_2, away_player_3, away_player_4,
               away_player_5, away_player_6, away_player_7, away_player_8,
               away_player_9, away_player_10, away_player_11
           ) AS lineup
    FROM main."Match"
    WHERE home_team_goal <> away_team_goal
), player_totals AS (
    SELECT l.outcome,
           j.value AS player_api_id,
           COUNT(DISTINCT l.match_api_id) AS match_count
    FROM team_lineups AS l
    CROSS JOIN json_each(l.lineup) AS j
    WHERE j.value IS NOT NULL
    GROUP BY l.outcome, j.value
), ranked AS (
    SELECT outcome, player_api_id, match_count,
           RANK() OVER (PARTITION BY outcome ORDER BY match_count DESC) AS outcome_rank
    FROM player_totals
)
SELECT r.outcome, p.player_name, r.match_count
FROM ranked AS r
JOIN main.Player AS p ON p.player_api_id = r.player_api_id
WHERE r.outcome_rank = 1
ORDER BY CASE r.outcome WHEN 'Win' THEN 1 ELSE 2 END, p.player_name
LIMIT 1000

WITH RECURSIVE
daily_changes AS (
    SELECT
        customer_id,
        txn_date AS balance_date,
        SUM(CASE WHEN txn_type = 'deposit' THEN txn_amount ELSE -txn_amount END) AS net_change
    FROM main.customer_transactions
    GROUP BY customer_id, txn_date
),
reporting_end AS (
    SELECT MAX(balance_date) AS last_day
    FROM daily_changes
),
customer_days(customer_id, balance_date, baseline_month, last_day) AS (
    SELECT
        d.customer_id,
        MIN(d.balance_date),
        strftime('%Y-%m', MIN(d.balance_date)),
        r.last_day
    FROM daily_changes AS d
    CROSS JOIN reporting_end AS r
    GROUP BY d.customer_id, r.last_day

    UNION ALL

    SELECT customer_id, date(balance_date, '+1 day'), baseline_month, last_day
    FROM customer_days
    WHERE balance_date < last_day
),
daily_balances AS (
    SELECT
        c.customer_id,
        c.balance_date,
        c.baseline_month,
        SUM(COALESCE(d.net_change, 0)) OVER (
            PARTITION BY c.customer_id
            ORDER BY c.balance_date
            ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
        ) AS running_balance
    FROM customer_days AS c
    LEFT JOIN daily_changes AS d
        ON d.customer_id = c.customer_id
       AND d.balance_date = c.balance_date
),
rolling_balances AS (
    SELECT
        customer_id,
        strftime('%Y-%m', balance_date) AS month,
        baseline_month,
        COUNT(*) OVER trailing_days AS days_in_window,
        AVG(running_balance) OVER trailing_days AS average_balance
    FROM daily_balances
    WINDOW trailing_days AS (
        PARTITION BY customer_id
        ORDER BY balance_date
        ROWS BETWEEN 29 PRECEDING AND CURRENT ROW
    )
),
customer_monthly_maxima AS (
    SELECT
        customer_id,
        month,
        MAX(MAX(average_balance, 0.0)) AS max_30_day_average_balance
    FROM rolling_balances
    WHERE days_in_window = 30
      AND month > baseline_month
    GROUP BY customer_id, month
)
SELECT
    month,
    SUM(max_30_day_average_balance) AS total_max_30_day_average_balance
FROM customer_monthly_maxima
GROUP BY month
ORDER BY month

WITH RECURSIVE
daily_changes AS (
    SELECT customer_id,
           txn_date AS balance_date,
           SUM(CASE WHEN txn_type = 'deposit' THEN txn_amount ELSE -txn_amount END) AS net_change
    FROM main.customer_transactions
    GROUP BY customer_id, txn_date
),
customer_starts AS (
    SELECT customer_id, MIN(balance_date) AS first_day
    FROM daily_changes
    GROUP BY customer_id
),
reporting_end AS (
    SELECT MAX(balance_date) AS last_day
    FROM daily_changes
),
customer_days(customer_id, balance_date, baseline_month) AS (
    SELECT customer_id, first_day, strftime('%Y-%m', first_day)
    FROM customer_starts

    UNION ALL

    SELECT customer_id, date(balance_date, '+1 day'), baseline_month
    FROM customer_days
    WHERE balance_date < (SELECT last_day FROM reporting_end)
),
daily_balances AS (
    SELECT c.customer_id,
           c.balance_date,
           c.baseline_month,
           SUM(COALESCE(d.net_change, 0)) OVER (
               PARTITION BY c.customer_id
               ORDER BY c.balance_date
               ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
           ) AS running_balance
    FROM customer_days AS c
    LEFT JOIN daily_changes AS d
        ON d.customer_id = c.customer_id
       AND d.balance_date = c.balance_date
),
rolling_balances AS (
    SELECT customer_id,
           strftime('%Y-%m', balance_date) AS month,
           baseline_month,
           COUNT(*) OVER trailing_30_days AS days_in_window,
           AVG(running_balance) OVER trailing_30_days AS average_balance
    FROM daily_balances
    WINDOW trailing_30_days AS (
        PARTITION BY customer_id
        ORDER BY balance_date
        ROWS BETWEEN 29 PRECEDING AND CURRENT ROW
    )
),
customer_monthly_maxima AS (
    SELECT customer_id,
           month,
           MAX(MAX(average_balance, 0.0)) AS max_30_day_average_balance
    FROM rolling_balances
    WHERE days_in_window = 30
      AND month > baseline_month
    GROUP BY customer_id, month
)
SELECT month,
       SUM(max_30_day_average_balance) AS total_max_30_day_average_balance
FROM customer_monthly_maxima
GROUP BY month
ORDER BY month
LIMIT 1000
