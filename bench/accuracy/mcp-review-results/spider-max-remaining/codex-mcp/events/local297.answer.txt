```sql
WITH monthly_nets AS (
    SELECT
        customer_id,
        DATE(txn_date, 'start of month') AS month_start,
        SUM(CASE txn_type
            WHEN 'deposit' THEN txn_amount
            WHEN 'withdrawal' THEN -txn_amount
        END) AS net_amount
    FROM main.customer_transactions
    WHERE txn_type IN ('deposit', 'withdrawal')
    GROUP BY customer_id, month_start
),
monthly_balances AS (
    SELECT
        customer_id,
        month_start,
        SUM(net_amount) OVER (
            PARTITION BY customer_id
            ORDER BY month_start
            ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
        ) AS closing_balance
    FROM monthly_nets
),
balance_history AS (
    SELECT
        closing_balance,
        LAG(closing_balance) OVER (
            PARTITION BY customer_id ORDER BY month_start
        ) AS prior_balance,
        ROW_NUMBER() OVER (
            PARTITION BY customer_id ORDER BY month_start DESC
        ) AS recency
    FROM monthly_balances
),
latest_growth AS (
    SELECT
        CASE
            WHEN prior_balance = 0 THEN 100.0 * closing_balance
            ELSE 100.0 * (closing_balance - prior_balance) / prior_balance
        END AS growth_pct
    FROM balance_history
    WHERE recency = 1
)
SELECT
    100.0 * COUNT(*) FILTER (WHERE growth_pct > 5) / NULLIF(COUNT(*), 0)
        AS percentage_of_customers
FROM latest_growth
LIMIT 1
```
