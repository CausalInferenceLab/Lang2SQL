```sql
WITH maxima AS (
    SELECT MAX(b.g) AS games_played,
           MAX(NULLIF(b.r, '')) AS runs,
           MAX(NULLIF(b.h, '')) AS hits,
           MAX(NULLIF(b.hr, '')) AS home_runs
    FROM main.batting AS b
),
leaders AS (
    SELECT 1 AS stat_order, 'Games played' AS statistic, b.player_id, b.g AS score
    FROM main.batting AS b
    JOIN maxima AS m ON b.g = m.games_played
    UNION ALL
    SELECT 2, 'Runs', b.player_id, b.r
    FROM main.batting AS b
    JOIN maxima AS m ON b.r = m.runs
    UNION ALL
    SELECT 3, 'Hits', b.player_id, b.h
    FROM main.batting AS b
    JOIN maxima AS m ON b.h = m.hits
    UNION ALL
    SELECT 4, 'Home runs', b.player_id, b.hr
    FROM main.batting AS b
    JOIN maxima AS m ON b.hr = m.home_runs
)
SELECT p.name_given, l.statistic, l.score
FROM leaders AS l
JOIN main.player AS p ON p.player_id = l.player_id
ORDER BY l.stat_order, p.name_given
LIMIT 1000
```
