```sql
WITH eligible AS (
    SELECT s.School, s.Phone, t.NumGE1500
    FROM main.schools AS s
    JOIN main.satscores AS t ON t.cds = s.CDSCode
    WHERE s.DOCType = 'Statewide Benefit Charter'
      AND t.rtype = 'S'
)
SELECT School, Phone
FROM eligible
WHERE NumGE1500 = (SELECT MAX(NumGE1500) FROM eligible)
ORDER BY School, Phone
LIMIT -1
```
