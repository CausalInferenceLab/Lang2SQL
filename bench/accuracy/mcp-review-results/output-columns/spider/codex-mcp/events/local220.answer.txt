```sql
WITH appearances AS (
    SELECT m.match_api_id,
           lineup.value AS player_api_id,
           CASE WHEN m.home_team_goal > m.away_team_goal THEN 'Win' ELSE 'Loss' END AS outcome
    FROM main."Match" AS m
    CROSS JOIN json_each(json_array(
        m.home_player_1, m.home_player_2, m.home_player_3, m.home_player_4,
        m.home_player_5, m.home_player_6, m.home_player_7, m.home_player_8,
        m.home_player_9, m.home_player_10, m.home_player_11
    )) AS lineup
    WHERE m.home_team_goal <> m.away_team_goal
      AND lineup.value IS NOT NULL

    UNION ALL

    SELECT m.match_api_id,
           lineup.value AS player_api_id,
           CASE WHEN m.away_team_goal > m.home_team_goal THEN 'Win' ELSE 'Loss' END AS outcome
    FROM main."Match" AS m
    CROSS JOIN json_each(json_array(
        m.away_player_1, m.away_player_2, m.away_player_3, m.away_player_4,
        m.away_player_5, m.away_player_6, m.away_player_7, m.away_player_8,
        m.away_player_9, m.away_player_10, m.away_player_11
    )) AS lineup
    WHERE m.home_team_goal <> m.away_team_goal
      AND lineup.value IS NOT NULL
),
match_counts AS (
    SELECT outcome, player_api_id, COUNT(DISTINCT match_api_id) AS match_count
    FROM appearances
    GROUP BY outcome, player_api_id
),
ranked AS (
    SELECT outcome, player_api_id,
           RANK() OVER (PARTITION BY outcome ORDER BY match_count DESC) AS outcome_rank
    FROM match_counts
)
SELECT r.outcome, p.player_name
FROM ranked AS r
JOIN main.Player AS p ON p.player_api_id = r.player_api_id
WHERE r.outcome_rank = 1
ORDER BY r.outcome DESC, p.player_name
LIMIT 1000
```
