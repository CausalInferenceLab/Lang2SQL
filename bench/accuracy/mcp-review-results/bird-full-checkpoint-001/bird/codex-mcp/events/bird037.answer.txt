```sql
WITH benchmark AS (
    SELECT MAX(t.AvgScrMath) AS highest_avg_sat_math
    FROM satscores AS t
    JOIN schools AS s ON s.CDSCode = t.cds
    WHERE t.rtype = 'S'
      AND s.StatusType = 'Active'
      AND t.NumTstTakr >= 10
)
SELECT
    s.School AS school_name,
    s.Phone AS phone,
    s.Website AS website,
    t.AvgScrRead AS avg_sat_reading,
    t.AvgScrMath AS avg_sat_math,
    t.AvgScrWrite AS avg_sat_writing,
    s.Charter AS is_charter,
    f."Enrollment (K-12)" AS k12_enrollment,
    ROUND(100.0 * f."FRPM Count (K-12)" / NULLIF(f."Enrollment (K-12)", 0), 2) AS frpm_eligible_pct,
    b.highest_avg_sat_math,
    b.highest_avg_sat_math - t.AvgScrMath AS math_points_below_highest
FROM schools AS s
JOIN satscores AS t ON t.cds = s.CDSCode
JOIN frpm AS f ON f.CDSCode = s.CDSCode
CROSS JOIN benchmark AS b
WHERE s.CDSCode = '07618040113118'
  AND f."Academic Year" = '2014-2015'
LIMIT 1
```
