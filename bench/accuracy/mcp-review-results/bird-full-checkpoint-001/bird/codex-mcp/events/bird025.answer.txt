```sql
SELECT ROUND(AVG(COALESCE(t.NumTstTakr, 0)), 2) AS avg_sat_test_takers_per_school
FROM main.schools AS s
LEFT JOIN main.satscores AS t ON t.cds = s.CDSCode
WHERE s.County = 'Los Angeles'
  AND s.School IS NOT NULL
  AND s.OpenDate >= '2007-01-01'
  AND s.OpenDate < '2008-01-01'
  AND s.DOC IN ('00', '52', '54', '56')
LIMIT 1
```
