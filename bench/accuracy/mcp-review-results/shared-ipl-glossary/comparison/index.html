WITH bowling_averages AS (
    SELECT
        b.bowler,
        1.0 * SUM(COALESCE(s.runs_scored, 0) + COALESCE(e.extra_runs, 0))
            / COUNT(w.player_out) AS bowling_average
    FROM main.ball_by_ball AS b
    LEFT JOIN main.batsman_scored AS s
        USING (match_id, innings_no, over_id, ball_id)
    LEFT JOIN main.extra_runs AS e
        ON (e.match_id, e.innings_no, e.over_id, e.ball_id)
         = (b.match_id, b.innings_no, b.over_id, b.ball_id)
        AND e.extra_type IN ('wides', 'noballs')
    LEFT JOIN main.wicket_taken AS w
        ON (w.match_id, w.innings_no, w.over_id, w.ball_id)
         = (b.match_id, b.innings_no, b.over_id, b.ball_id)
        AND w.kind_out IN ('bowled', 'caught', 'caught and bowled', 'hit wicket', 'lbw', 'stumped')
    GROUP BY b.bowler
    HAVING COUNT(w.player_out) > 0
)
SELECT p.player_name
FROM bowling_averages AS a
JOIN main.player AS p ON p.player_id = a.bowler
WHERE a.bowling_average = (SELECT MIN(bowling_average) FROM bowling_averages)
LIMIT 1000

WITH season_matches AS (
    SELECT match_id
    FROM main."match"
    WHERE season_id = 5

appearances AS (
    SELECT pm.player_id, COUNT(*) AS matches_played

    SELECT b.striker AS player_id, SUM(s.runs_scored) AS total_runs
    FROM season_matches AS m
    JOIN main.ball_by_ball AS b ON b.match_id = m.match_id
    JOIN main.batsman_scored AS s USING (match_id, innings_no, over_id, ball_id)
    GROUP BY b.striker
),
dismissals AS (
    SELECT w.player_out AS player_id, COUNT(*) AS times_out
    FROM main.wicket_taken AS w
    JOIN season_matches AS m ON m.match_id = w.match_id

SELECT p.player_name,

    SELECT match_id, innings_no, over_id, runs_scored AS runs
    FROM main.batsman_scored

               PARTITION BY match_id

               ORDER BY SUM(runs) DESC, innings_no, over_id

    FROM scoring

WITH run_entries AS (
    SELECT match_id, innings_no, over_id, runs_scored AS runs
    FROM main.batsman_scored
    UNION ALL
    SELECT match_id, innings_no, over_id, extra_runs AS runs
    FROM main.extra_runs
),
over_totals AS (
    SELECT match_id, innings_no, over_id, SUM(runs) AS total_runs
    FROM run_entries
    GROUP BY match_id, innings_no, over_id
),
ranked_overs AS (
    SELECT over_totals.*,
           ROW_NUMBER() OVER (
               PARTITION BY match_id
               ORDER BY total_runs DESC, innings_no, over_id
           ) AS over_rank
    FROM over_totals
),
selected_bowlers AS (
    SELECT DISTINCT r.match_id, r.innings_no, r.over_id, r.total_runs, b.bowler
    FROM ranked_overs AS r
    LEFT JOIN main.ball_by_ball AS b USING (match_id, innings_no, over_id)
    WHERE r.over_rank = 1
)
SELECT match_id, innings_no, over_id,
       GROUP_CONCAT(bowler, ', ' ORDER BY bowler) AS bowler_ids,
       total_runs, AVG(total_runs) OVER () AS average_highest_over_runs
FROM selected_bowlers
GROUP BY match_id, innings_no, over_id, total_runs
ORDER BY match_id
LIMIT 1000
