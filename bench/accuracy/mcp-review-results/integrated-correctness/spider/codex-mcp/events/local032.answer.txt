```sql
WITH seller_orders AS (
    SELECT i.seller_id,
           o.order_id,
           o.customer_id,
           SUM(i.price - i.freight_value) AS order_profit
    FROM main.olist_orders AS o
    JOIN main.olist_order_items AS i ON i.order_id = o.order_id
    WHERE o.order_status = 'delivered'
    GROUP BY i.seller_id, o.order_id, o.customer_id
),
five_star_reviews AS (
    SELECT order_id, COUNT(*) AS rating_count
    FROM main.olist_order_reviews
    WHERE review_score = 5
    GROUP BY order_id
),
seller_metrics AS (
    SELECT so.seller_id,
           COUNT(DISTINCT c.customer_unique_id) AS customer_count,
           ROUND(SUM(so.order_profit), 2) AS profit,
           COUNT(*) AS order_count,
           SUM(COALESCE(r.rating_count, 0)) AS five_star_count
    FROM seller_orders AS so
    LEFT JOIN main.olist_customers AS c ON c.customer_id = so.customer_id
    LEFT JOIN five_star_reviews AS r ON r.order_id = so.order_id
    GROUP BY so.seller_id
),
achievements AS (
    SELECT 1 AS category_order, 'Most distinct customer unique IDs' AS achievement,
           seller_id, customer_count AS value
    FROM seller_metrics
    UNION ALL
    SELECT 2, 'Highest total profit', seller_id, profit
    FROM seller_metrics
    UNION ALL
    SELECT 3, 'Most distinct orders', seller_id, order_count
    FROM seller_metrics
    UNION ALL
    SELECT 4, 'Most 5-star ratings', seller_id, five_star_count
    FROM seller_metrics
),
ranked AS (
    SELECT category_order, achievement, seller_id, value,
           ROW_NUMBER() OVER (
               PARTITION BY category_order
               ORDER BY value DESC, seller_id
           ) AS position
    FROM achievements
)
SELECT achievement, seller_id, value
FROM ranked
WHERE position = 1
ORDER BY category_order
LIMIT 4
```
