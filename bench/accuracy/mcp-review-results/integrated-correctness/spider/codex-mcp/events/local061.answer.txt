```sql
WITH monthly_sales AS (
    SELECT s.prod_id,
           t.calendar_month_number AS month,
           t.calendar_year AS sales_year,
           SUM(s.amount_sold) AS sales_amount
    FROM main.sales AS s
    JOIN main.customers AS cu ON cu.cust_id = s.cust_id
    JOIN main.countries AS co ON co.country_id = cu.country_id
    JOIN main.times AS t ON t.time_id = s.time_id
    JOIN main.promotions AS p ON p.promo_id = s.promo_id
    JOIN main.channels AS ch ON ch.channel_id = s.channel_id
    WHERE co.country_name = 'France'
      AND p.promo_total_id = 1
      AND ch.channel_total_id = 1
      AND t.calendar_year IN (2019, 2020)
    GROUP BY s.prod_id, t.calendar_month_number, t.calendar_year
),
projected_sales AS (
    SELECT current_sales.month,
           ((current_sales.sales_amount - prior_sales.sales_amount)
            / NULLIF(prior_sales.sales_amount, 0))
            * current_sales.sales_amount + current_sales.sales_amount AS projected_amount
    FROM monthly_sales AS current_sales
    JOIN monthly_sales AS prior_sales
      ON prior_sales.prod_id = current_sales.prod_id
     AND prior_sales.month = current_sales.month
     AND prior_sales.sales_year = 2019
    WHERE current_sales.sales_year = 2020
)
SELECT ps.month,
       ROUND(AVG(ps.projected_amount * COALESCE(fx.to_us, 1.0)), 2)
           AS avg_projected_sales_usd
FROM projected_sales AS ps
LEFT JOIN main.currency AS fx
  ON fx.country = 'France'
 AND fx.year = 2021
 AND fx.month = ps.month
GROUP BY ps.month
ORDER BY ps.month
LIMIT 12
```
