WITH recorded_races AS (
    SELECT r.race_id, r.year, r.round
    FROM main.races AS r
    WHERE r.race_id IN (SELECT race_id FROM main.results)
),
appearances AS (
    SELECT r.year, r.round, s.driver_id, s.constructor_id
    FROM recorded_races AS r
    JOIN main.results AS s ON s.race_id = r.race_id
),
driver_seasons AS (
    SELECT DISTINCT year, driver_id
    FROM appearances
),
missed_rounds AS (
    SELECT d.year, d.driver_id,
           MIN(r.round) AS first_missed_round,
           MAX(r.round) AS last_missed_round
    FROM driver_seasons AS d
    JOIN recorded_races AS r ON r.year = d.year
    LEFT JOIN appearances AS a
      ON a.year = d.year
     AND a.driver_id = d.driver_id
     AND a.round = r.round
    WHERE a.driver_id IS NULL
    GROUP BY d.year, d.driver_id
    HAVING COUNT(*) < 3
),
team_changes AS (
    SELECT DISTINCT m.year, m.driver_id, m.first_missed_round, m.last_missed_round
    FROM missed_rounds AS m
    JOIN appearances AS before_gap
      ON before_gap.year = m.year
     AND before_gap.driver_id = m.driver_id
     AND before_gap.round = m.first_missed_round - 1
    JOIN appearances AS after_gap
      ON after_gap.year = m.year
     AND after_gap.driver_id = m.driver_id
     AND after_gap.round = m.last_missed_round + 1
    WHERE before_gap.constructor_id <> after_gap.constructor_id
)
SELECT AVG(first_missed_round) AS avg_first_missed_round,
       AVG(last_missed_round) AS avg_last_missed_round
FROM team_changes
LIMIT 1

    SELECT ol.id AS order_line_id,
           ol.product_id,
           ol.qty,
           SUM(ol.qty) OVER (
               PARTITION BY ol.product_id
               ORDER BY ol.id
               ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
           ) AS demand_end
    FROM main.orderlines AS ol
    WHERE ol.order_id = 423
),
warehouse_stock AS (
    SELECT i.product_id,
           l.aisle,
           l.position,
           i.qty,
           SUM(i.qty) OVER (
               PARTITION BY i.product_id
               ORDER BY p.purchased, i.qty, i.id
               ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
           ) AS stock_end
    FROM main.inventory AS i
    JOIN main.locations AS l ON l.id = i.location_id
    JOIN main.purchases AS p ON p.id = i.purchase_id
    WHERE l.warehouse = 1
      AND i.qty > 0
      AND i.product_id IN (SELECT product_id FROM order_demand)
)
SELECT d.order_line_id,
       d.product_id,
       s.aisle,
       s.position,
       SUM(
           MIN(d.demand_end, s.stock_end)
           - MAX(d.demand_end - d.qty, s.stock_end - s.qty)
       ) AS pick_qty
FROM order_demand AS d
JOIN warehouse_stock AS s
    ON s.product_id = d.product_id
   AND s.stock_end - s.qty < d.demand_end
   AND d.demand_end - d.qty < s.stock_end
WHERE d.qty > 0
GROUP BY d.order_line_id, d.product_id, s.aisle, s.position
ORDER BY d.order_line_id, MIN(s.stock_end)
LIMIT 1000

WITH demand AS (
    SELECT ol.id AS order_line_id, ol.product_id, ol.qty,
           SUM(ol.qty) OVER (
               PARTITION BY ol.product_id
               ORDER BY ol.id
               ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
           ) AS demand_end
    FROM main.orderlines AS ol
    WHERE ol.order_id = 423
),
supply AS (
    SELECT i.product_id, l.aisle, l.position, i.qty,
           SUM(i.qty) OVER (
               PARTITION BY i.product_id
               ORDER BY p.purchased, i.qty, i.id
               ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
           ) AS supply_end
    FROM main.inventory AS i
    JOIN main.locations AS l ON l.id = i.location_id
    JOIN main.purchases AS p ON p.id = i.purchase_id
    WHERE l.warehouse = 1
      AND i.qty > 0
      AND i.product_id IN (SELECT product_id FROM demand)
)
SELECT d.order_line_id, d.product_id, s.aisle, s.position,
       MIN(d.demand_end, s.supply_end)
         - MAX(d.demand_end - d.qty, s.supply_end - s.qty) AS pick_qty
FROM demand AS d
JOIN supply AS s
  ON s.product_id = d.product_id
 AND s.supply_end > d.demand_end - d.qty
 AND d.demand_end > s.supply_end - s.qty
ORDER BY d.order_line_id, s.supply_end
LIMIT 1000

           json_extract(city, '$.en') AS city,
           radians(CAST(substr(coordinates, 2, instr(coordinates, ',') - 2) AS REAL)) AS longitude,

    SELECT MIN(departure.city, arrival.city) AS city1,
           MAX(departure.city, arrival.city) AS city2,

               pow(sin((arrival.latitude - departure.latitude) / 2.0), 2)
               + cos(departure.latitude) * cos(arrival.latitude)
                 * pow(sin((arrival.longitude - departure.longitude) / 2.0), 2)
           ))) AS distance_km
    FROM main.flights AS flight
