```sql
WITH type_counts AS (
    SELECT c.types AS card_type, COUNT(*) AS card_count
    FROM cards AS c
    WHERE c.watermark = 'boros'
      AND c.uuid IN (
          SELECT fd.uuid
          FROM foreign_data AS fd
          WHERE fd.language <> 'English'
            AND fd.language <> ''
      )
    GROUP BY c.types
)
SELECT card_type, card_count
FROM type_counts
WHERE card_count = (SELECT MAX(card_count) FROM type_counts)
ORDER BY card_type
LIMIT 1000
```
