WITH type_counts AS (
    SELECT c.types AS card_type, COUNT(*) AS card_count
    FROM cards AS c
    WHERE c.watermark = 'boros'
      AND c.uuid IN (
          SELECT f.uuid
          FROM foreign_data AS f
          WHERE f.language <> 'English'
      )
    GROUP BY c.types
)
SELECT card_type, card_count
FROM type_counts
WHERE card_count = (SELECT MAX(card_count) FROM type_counts)
ORDER BY card_type
LIMIT 1000

SELECT
    f.name AS japanese_name,
    c.type AS english_card_type,
    COUNT(DISTINCT r.id) AS ruling_total
FROM main.foreign_data AS f
JOIN main.cards AS c ON c.uuid = f.uuid
LEFT JOIN main.rulings AS r ON r.uuid = c.uuid
WHERE f.language = 'Japanese'
GROUP BY f.name, c.type
HAVING MAX(CASE WHEN c.watermark = 'boros' THEN 1 ELSE 0 END) = 1
ORDER BY ruling_total DESC, f.name, c.type
LIMIT 1000

WITH editor_counts AS (
    SELECT h.UserId, COUNT(DISTINCT h.RevisionGUID) AS edit_comment_count
    FROM posts AS p
    JOIN postHistory AS h ON h.PostId = p.Id
    WHERE p.Title = 'Why square the difference instead of taking the absolute value in standard deviation?'
      AND TRIM(h.Comment) <> ''
    GROUP BY h.UserId
)
SELECT u.DisplayName, e.edit_comment_count
FROM editor_counts AS e
JOIN users AS u ON u.Id = e.UserId
WHERE e.edit_comment_count = (SELECT MAX(edit_comment_count) FROM editor_counts)
ORDER BY u.DisplayName
LIMIT 1000
